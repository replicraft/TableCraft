def pin(path):
    data = path.read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def u16(data, offset):
    return struct.unpack_from('<H', data, offset)[0]

def u32(data, offset):
    return struct.unpack_from('<I', data, offset)[0]

def be32(data, offset):
    return struct.unpack_from('>I', data, offset)[0]

def row(raw, reference):
    number, slot = reference >> 8, reference & 255
    page = raw[number * 4096:(number + 1) * 4096]
    assert page[:2] == b'\1\1' and slot < u16(page, 12)
    word = u16(page, 14 + 2 * slot)
    assert word & 0xe000 == 0
    start = word & 0x1fff
    end = 4096 if slot == 0 else u16(page, 12 + 2 * slot) & 0x1fff
    assert 14 + 2 * u16(page, 12) <= start < end <= 4096
    return page[start:end]

def members(raw, reference):
    payload = row(raw, reference)
    if payload[0] == 0:
        base = u32(payload, 1)
        result = [base + byte * 8 + bit for byte, mask in enumerate(payload[5:])
                  for bit in range(8) if mask & 1 << bit]
    elif payload[0] == 1:
        assert (len(payload) - 1) % 4 == 0
        result = []
        for number in range((len(payload) - 1) // 4):
            pointer = u32(payload, 1 + number * 4)
            if pointer:
                page = raw[pointer * 4096:(pointer + 1) * 4096]
                assert page[:4] == bytes([5, 1, 0, 0])
                result.extend(number * 32736 + byte * 8 + bit
                              for byte, mask in enumerate(page[4:])
                              for bit in range(8) if mask & 1 << bit)
    else:
        raise AssertionError('unknown map')
    assert result == sorted(set(result)) and all(page < len(raw) // 4096 for page in result)
    return result

def physical(raw, definition, number):
    page = raw[definition * 4096:(definition + 1) * 4096]
    columns = u16(page, 45)
    physical_count = u32(page, 51)
    assert number < physical_count
    offset = 63 + physical_count * 12 + columns * 25
    for _ in range(columns):
        length = u16(page, offset)
        offset += 2 + length
    declaration = offset + number * 52
    return u32(page, declaration + 34), u32(page, declaration + 38)

def decode_tree(raw, number, owner, visited=None):
    visited = set() if visited is None else visited
    assert number not in visited
    visited.add(number)
    page = raw[number * 4096:(number + 1) * 4096]
    assert page[0] in [3, 4] and page[1] == 1 and u32(page, 4) == owner
    leaf = page[0] == 4
    ends = [byte * 8 + bit for byte, mask in enumerate(page[27:480])
            for bit in range(8) if mask & 1 << bit]
    prefix = u16(page, 24)
    previous = 0
    first = None
    entries = []
    leaves = []
    nodes = []
    children = []
    for end in ends:
        assert previous < end <= 3616
        fragment = page[480 + previous:480 + end]
        full = fragment if first is None else first[:prefix] + fragment
        if first is None:
            first = full
        assert prefix <= len(first)
        trailer = 4 if leaf else 8
        assert len(full) > trailer
        key = full[:-trailer]
        reference = be32(full, len(full) - trailer)
        if leaf:
            entries.append((key.hex(), reference >> 8, reference & 255))
        else:
            child = be32(full, len(full) - 4)
            children.append(child)
        previous = end
    assert u16(page, 2) == 3616 - previous
    assert ends or prefix == 0
    links = {'page': number, 'kind': page[0], 'height': page[26],
             'previous': u32(page, 12), 'next': u32(page, 16), 'tail': u32(page, 20)}
    if leaf:
        assert links['tail'] == 0
        leaves.append(links | {'entries': len(entries)})
    else:
        if links['tail']:
            children.append(links['tail'])
        nodes.append(links | {'children': children[:]})
        for child in children:
            keys, child_leaves, child_nodes = decode_tree(raw, child, owner, visited)
            entries.extend(keys)
            leaves.extend(child_leaves)
            nodes.extend(child_nodes)
    return entries, leaves, nodes

original_pins = [pin(path) for path in FILES]
commands = []
results = []
for path in FILES:
    output = WORK / (path.stem + '.getters.tsv')
    errors = WORK / (path.stem + '.getters.stderr')
    command = ['java', '-cp', f'{JAR}:{WORK}',
               'com.healthmarketscience.jackcess.impl.EncodedDeleteFixture', 'oracle', str(path)]
    if not output.exists():
        with output.open('xb') as stdout, errors.open('xb') as stderr:
            completed = subprocess.run(command, stdout=stdout, stderr=stderr, timeout=45)
        assert completed.returncode == 0, (path, completed.returncode)
    commands.append({'argv': command, 'exit_code': 0, 'stdout': pin(output), 'stderr': pin(errors)})
    raw = path.read_bytes()
    tables = {}
    indexes = []
    keys = {}
    payload = []
    for line in output.read_text().splitlines():
        fields = line.split('\t')
        if fields[0] == 'TABLE':
            tables[fields[1]] = {'rows': int(fields[2]), 'definition': int(fields[3]), 'getter_rows': []}
        elif fields[0] == 'ROW':
            tables[fields[1]]['getter_rows'].append(fields[2:])
        elif fields[0] == 'COLUMN':
            tables[fields[1]]['column'] = fields[2:]
        elif fields[0] == 'PAYLOAD':
            payload.append(fields[1:])
        elif fields[0] == 'INDEX':
            indexes.append({'table': fields[1], 'name': fields[2], 'physical': int(fields[3]),
                            'root': int(fields[4]), 'kind': int(fields[5]), 'height': int(fields[6]),
                            'owned_count': int(fields[7]), 'historical_count': int(fields[8]),
                            'counter_offset': int(fields[9]), 'entries': int(fields[10])})
        elif fields[0] == 'KEY':
            keys.setdefault((fields[1], fields[2]), []).append((fields[5], int(fields[3]), int(fields[4])))
    for index in indexes:
        table = tables[index['table']]
        usage, root = physical(raw, table['definition'], index['physical'])
        assert root == index['root']
        entries, leaves, nodes = decode_tree(raw, root, table['definition'])
        assert entries == keys.get((index['table'], index['name']), [])
        assert len(entries) == index['entries'] == table['rows']
        owned = members(raw, usage)
        assert len(owned) == index['owned_count']
        traversed = [item['page'] for item in leaves + nodes]
        assert set(traversed).issubset(owned)
        index.update({'usage': usage, 'owned': owned, 'leaves': leaves, 'nodes': nodes})
    for name, table in tables.items():
        if name != 'PreservedPayloads':
            assert len(table['getter_rows']) == table['rows']
    results.append({'input': pin(path), 'header_code': raw[20], 'subversion': raw[21],
                    'tables': tables, 'indexes': indexes, 'preserved_payload_hashes': payload,
                    'raw_getter_keys_match': True})
    print(path.name, 'rows', sum(table['rows'] for table in tables.values()), 'indexes', len(indexes), flush=True)
